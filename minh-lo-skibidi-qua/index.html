import turtle t = turtle.Turtle() t.speed(3) # ====== ĐẦU ====== def draw_head(): t.penup() t.goto(0, 100) t.pendown() t.circle(40) # ====== MẮT ====== def draw_eyes(): # mắt trái t.penup() t.goto(-15, 130) t.pendown() t.dot(8) # mắt phải t.penup() t.goto(15, 130) t.pendown() t.dot(8) # ====== MIỆNG ====== def draw_mouth(): t.penup() t.goto(-10, 110) t.setheading(-60) t.pendown() t.circle(20, 120) # ====== TÓC ====== def draw_hair(): t.penup() t.goto(-40, 140) t.pendown() t.setheading(0) t.forward(80) t.right(90) t.forward(20) t.right(90) t.forward(80) # ====== THÂN ====== def draw_body(): t.penup() t.goto(0, 100) t.setheading(-90) t.pendown() t.forward(120) # ====== TAY ====== def draw_arms(): t.penup() t.goto(0, 60) t.pendown() # tay trái t.setheading(210) t.forward(60) # quay lại t.penup() t.goto(0, 60) t.pendown() # tay phải t.setheading(-30) t.forward(60) # ====== CHÂN ====== def draw_legs(): t.penup() t.goto(0, -20) t.pendown() # chân trái t.setheading(240) t.forward(80) t.penup() t.goto(0, -20) t.pendown() # chân phải t.setheading(-60) t.forward(80) # ====== VẼ ====== draw_head() draw_eyes() draw_mouth() draw_hair() draw_body() draw_arms() draw_legs() turtle.done()
